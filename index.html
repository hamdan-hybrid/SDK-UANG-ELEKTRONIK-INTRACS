Parking API

Check balance cURL
curl -X POST http://localhost:4173/api/parking/check-balance \
  -H "Content-Type: application/json" \
  -d '{"transactionId":"PARK-0001","amount":5000}'
Deduct cURL
curl -X POST http://localhost:4173/api/parking/deduct \
  -H "Content-Type: application/json" \
  -d '{"transactionId":"PARK-0001","amount":5000,"gateId":"GATE-01"}'
PHP
<?php
$data = [
  'transactionId' => 'PARK-0001',
  'amount' => 5000,
  'gateId' => 'GATE-01'
];
$ch = curl_init('http://localhost:4173/api/parking/deduct');
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
$response = json_decode(curl_exec($ch), true);
curl_close($ch);
if ($response['success'] === true) { /* buka palang */ }

{
  "success": true,
  "transactionId": "PARK-0001",
